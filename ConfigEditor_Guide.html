Overview

The ConfigEditor is a comprehensive configuration management and persistence engine that serves as the central configuration hub for the entire Beep platform. It handles all aspects of application configuration including data connections, drivers, entity structures, workflows, mappings, and various application settings with robust persistence capabilities.

?? Configuration Management

ConfigEditor manages every aspect of the application's configuration from data source connections to workflow definitions, providing a unified interface for configuration persistence, validation, and retrieval across all platform components.

?? Initialization Steps

1. Platform Path Resolution: Determine appropriate config path for the OS

2. Configuration Loading: Load main configuration from JSON files

3. Directory Creation: Ensure all required directories exist

4. Connection Loading: Load data connections and drivers

5. Component Initialization: Initialize workflows, mappings, and other components

6. Validation: Validate configuration integrity and paths

Configuration Migration and Backup

// Configuration backup and migration utilities public class ConfigurationManager { private readonly ConfigEditor configEditor; public ConfigurationManager(ConfigEditor editor) { configEditor = editor; } public async Task BackupConfiguration(string backupPath) { try { var backupDir = Path.Combine(backupPath, $"Config_Backup_{DateTime.Now:yyyyMMdd_HHmmss}"); Directory.CreateDirectory(backupDir); // Backup main configuration configEditor.SaveConfigValues(); File.Copy( Path.Combine(configEditor.ConfigPath, "Config.json"), Path.Combine(backupDir, "Config.json") ); // Backup data connections configEditor.SaveDataconnectionsValues(); File.Copy( Path.Combine(configEditor.ConfigPath, "DataConnections.json"), Path.Combine(backupDir, "DataConnections.json") ); // Backup drivers configEditor.SaveConnectionDriversConfigValues(); File.Copy( Path.Combine(configEditor.ConfigPath, "ConnectionConfig.json"), Path.Combine(backupDir, "ConnectionConfig.json") ); // Backup workflows configEditor.SaveWork(); File.Copy( Path.Combine(configEditor.Config.WorkFlowPath, "DataWorkFlow.json"), Path.Combine(backupDir, "DataWorkFlow.json") ); // Backup entity structures var entitiesBackupDir = Path.Combine(backupDir, "Entities"); Directory.CreateDirectory(entitiesBackupDir); foreach (var file in Directory.GetFiles(configEditor.Config.EntitiesPath, "*_ES.json")) { var fileName = Path.GetFileName(file); File.Copy(file, Path.Combine(entitiesBackupDir, fileName)); } Console.WriteLine($"? Configuration backed up to: {backupDir}"); } catch (Exception ex) { Console.WriteLine($"? Backup failed: {ex.Message}"); } } public async Task RestoreConfiguration(string backupPath) { try { // Restore main configuration if (File.Exists(Path.Combine(backupPath, "Config.json"))) { File.Copy( Path.Combine(backupPath, "Config.json"), Path.Combine(configEditor.ConfigPath, "Config.json"), overwrite: true ); configEditor.LoadConfigValues(); } // Restore data connections if (File.Exists(Path.Combine(backupPath, "DataConnections.json"))) { File.Copy( Path.Combine(backupPath, "DataConnections.json"), Path.Combine(configEditor.ConfigPath, "DataConnections.json"), overwrite: true ); configEditor.LoadDataConnectionsValues(); } // Restore drivers if (File.Exists(Path.Combine(backupPath, "ConnectionConfig.json"))) { File.Copy( Path.Combine(backupPath, "ConnectionConfig.json"), Path.Combine(configEditor.ConfigPath, "ConnectionConfig.json"), overwrite: true ); configEditor.LoadConnectionDriversConfigValues(); } // Restore workflows if (File.Exists(Path.Combine(backupPath, "DataWorkFlow.json"))) { File.Copy( Path.Combine(backupPath, "DataWorkFlow.json"), Path.Combine(configEditor.Config.WorkFlowPath, "DataWorkFlow.json"), overwrite: true ); configEditor.ReadWork(); } // Restore entity structures var entitiesBackupDir = Path.Combine(backupPath, "Entities"); if (Directory.Exists(entitiesBackupDir)) { foreach (var file in Directory.GetFiles(entitiesBackupDir, "*_ES.json")) { var fileName = Path.GetFileName(file); File.Copy(file, Path.Combine(configEditor.Config.EntitiesPath, fileName), overwrite: true); } } Console.WriteLine("? Configuration restored successfully"); } catch (Exception ex) { Console.WriteLine($"? Restore failed: {ex.Message}"); } } public void ValidateConfiguration() { var issues = new List<string>(); // Validate data connections foreach (var connection in configEditor.DataConnections) { if (string.IsNullOrEmpty(connection.ConnectionName)) issues.Add($"Connection has empty name: {connection.GuidID}"); if (connection.Category == DatasourceCategory.FILE) { if (string.IsNullOrEmpty(connection.FilePath) || string.IsNullOrEmpty(connection.FileName)) issues.Add($"File connection missing path/filename: {connection.ConnectionName}"); } else { if (string.IsNullOrEmpty(connection.ConnectionString)) issues.Add($"Connection missing connection string: {connection.ConnectionName}"); } } // Validate drivers foreach (var driver in configEditor.DataDriversClasses) { if (string.IsNullOrEmpty(driver.PackageName)) issues.Add($"Driver missing package name: {driver.GuidID}"); if (string.IsNullOrEmpty(driver.DriverClass)) issues.Add($"Driver missing class name: {driver.PackageName}"); } // Validate workflows foreach (var workflow in configEditor.WorkFlows) { if (string.IsNullOrEmpty(workflow.DataWorkFlowName)) issues.Add($"Workflow missing name: {workflow.GuidID}"); if (workflow.Datasteps == null || workflow.Datasteps.Count == 0) issues.Add($"Workflow has no steps: {workflow.DataWorkFlowName}"); } if (issues.Count > 0) { Console.WriteLine("?? Configuration validation issues found:"); foreach (var issue in issues) { Console.WriteLine($" - {issue}"); } } else { Console.WriteLine("? Configuration validation passed"); } } }
?? Best Practice: Always validate configuration after initialization and implement backup strategies for production environments. Use the configuration manager utilities for reliable configuration management.
?? Important: ConfigEditor manages critical application settings. Always backup configurations before making significant changes and test configuration changes in development environments first.
