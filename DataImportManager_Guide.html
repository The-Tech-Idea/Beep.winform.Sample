Overview

The DataImportManager is a sophisticated ETL (Extract, Transform, Load) engine designed for high-performance data import operations. It provides a comprehensive solution for moving data between different data sources with advanced features like field mapping, transformation, batch processing, and real-time progress reporting.

?? ETL Operations

DataImportManager orchestrates the entire ETL pipeline: extracting data from source systems, applying transformations and mappings, and loading data into destination systems with validation and error handling throughout the process.

Basic Import Operation

// Basic data import setup var importManager = new DataImportManager(dmeEditor); // Configure source and destination importManager.SourceDataSourceName = "SourceDB"; importManager.SourceEntityName = "Customers"; importManager.DestDataSourceName = "DestinationDB"; importManager.DestEntityName = "ImportedCustomers"; // Load destination entity structure var loadResult = importManager.LoadDestEntityStructure("ImportedCustomers", "DestinationDB"); if (loadResult.Flag == Errors.Ok) { // Setup progress reporting var progress = new Progress<IPassedArgs>(args => { Console.WriteLine($"Progress: {args.Messege}"); }); // Run the import var importResult = await importManager.RunImportAsync(progress, CancellationToken.None); if (importResult.Flag == Errors.Ok) { Console.WriteLine("Import completed successfully"); } }

Advanced Import with Filtering and Mapping

// Advanced import configuration var importManager = new DataImportManager(dmeEditor); // Configure sources importManager.SourceDataSourceName = "CrmDB"; importManager.SourceEntityName = "Contacts"; importManager.DestDataSourceName = "DataWarehouse"; importManager.DestEntityName = "CustomerContacts"; // Add source filters importManager.SourceFilters = new List<AppFilter> { new AppFilter { FieldName = "Status", Operator = "=", FilterValue = "Active" }, new AppFilter { FieldName = "LastModified", Operator = ">=", FilterValue = DateTime.Now.AddDays(-30) } }; // Select specific fields importManager.SelectedFields = new List<string> { "ContactId", "FirstName", "LastName", "Email", "Phone", "Company" }; // Configure field mapping importManager.Mapping = new EntityDataMap { MappedEntities = new List<EntityMapping> { new EntityMapping { EntityName = "CustomerContacts", FieldMappings = new List<FieldMapping> { new FieldMapping { SourceField = "ContactId", DestinationField = "Id" }, new FieldMapping { SourceField = "FirstName", DestinationField = "First_Name" }, new FieldMapping { SourceField = "LastName", DestinationField = "Last_Name" }, new FieldMapping { SourceField = "Company", DestinationField = "Company_Name" } } } } }; // Load destination structure await Task.Run(() => importManager.LoadDestEntityStructure("CustomerContacts", "DataWarehouse")); // Setup advanced progress reporting var progress = new Progress<IPassedArgs>(args => { var percentage = (args.ParameterInt1 * 100) / totalRecords; Console.WriteLine($"Progress: {percentage}% - {args.Messege}"); // Update UI progress bar UpdateProgressBar(percentage); }); // Run import with large batch size for better performance var result = await importManager.RunImportAsync( progress, cancellationToken, batchSize: 500 );

Import with Custom Transformation

// Import with business logic transformation var importManager = new DataImportManager(dmeEditor); // Configure basic settings importManager.SourceDataSourceName = "SalesDB"; importManager.SourceEntityName = "RawLeads"; importManager.DestDataSourceName = "ProcessedDB"; importManager.DestEntityName = "QualifiedLeads"; // Load destination structure importManager.LoadDestEntityStructure("QualifiedLeads", "ProcessedDB"); // Define custom transformation logic Func<object, object> businessTransformation = (record) => { if (record is Dictionary<string, object> data) { // Apply lead scoring algorithm var score = CalculateLeadScore(data); data["LeadScore"] = score; // Qualify leads based on score data["Status"] = score > 70 ? "Qualified" : "Nurture"; // Standardize phone numbers if (data.ContainsKey("Phone")) { data["Phone"] = StandardizePhoneNumber(data["Phone"]?.ToString()); } // Set import metadata data["ImportedDate"] = DateTime.Now; data["ImportedBy"] = Environment.UserName; // Clean up data foreach (var key in data.Keys.ToList()) { if (data[key] == null || data[key].ToString().Trim() == "") { data[key] = DBNull.Value; } } } return record; }; // Execute import with transformation var progress = new Progress<IPassedArgs>(args => { Console.WriteLine($"Processed {args.ParameterInt1} leads: {args.Messege}"); }); var result = await importManager.RunImportAsync( progress, CancellationToken.None, transformation: businessTransformation, batchSize: 100 ); // Review import logs Console.WriteLine($"Import completed. Error Handling and Recovery

// Comprehensive error handling example public async Task RobustImportWithErrorHandling() { var importManager = new DataImportManager(dmeEditor); try { // Configure import importManager.SourceDataSourceName = "UnreliableSource"; importManager.SourceEntityName = "ProblematicData"; importManager.DestDataSourceName = "SafeDestination"; importManager.DestEntityName = "CleanedData"; // Load and validate destination var structureResult = importManager.LoadDestEntityStructure("CleanedData", "SafeDestination"); if (structureResult.Flag == Errors.Failed) { Console.WriteLine($"Failed to load destination structure: {structureResult.Message}"); return; } // Setup cancellation with timeout using var cts = new CancellationTokenSource(TimeSpan.FromMinutes(30)); // Setup detailed progress tracking var progress = new Progress<IPassedArgs>(args => { Console.WriteLine($"[{DateTime.Now:HH:mm:ss}] {args.Messege}"); // Log progress to file File.AppendAllText("import_progress.log", $"{DateTime.Now:yyyy-MM-dd HH:mm:ss} - {args.Messege}\n"); }); // Run import with smaller batches for better error isolation var result = await importManager.RunImportAsync( progress, cts.Token, batchSize: 50 ); // Handle results if (result.Flag == Errors.Ok) { Console.WriteLine("? Import completed successfully"); // Save detailed logs SaveImportReport(importManager.ImportLogData); } else { Console.WriteLine($"? Import failed: {result.Message}"); // Analyze errors var errorLogs = importManager.ImportLogData .Where(log => log.Message.Contains("Error") || log.Message.Contains("Failed")) .ToList(); Console.WriteLine($"Found {errorLogs.Count} error entries"); foreach (var error in errorLogs) { Console.WriteLine($" {error.Timestamp}: {error.Message}"); } } } catch (OperationCanceledException) { Console.WriteLine("?? Import was cancelled due to timeout or user request"); } catch (Exception ex) { Console.WriteLine($"?? Unexpected error during import: {ex.Message}"); // Log full exception details File.AppendAllText("import_errors.log", $"{DateTime.Now:yyyy-MM-dd HH:mm:ss} - EXCEPTION: {ex}\n"); } finally { // Cleanup and final reporting Console.WriteLine($"Import session completed. Check logs for details."); } } private void SaveImportReport(List<Importlogdata> logs) { var report = new { ImportSession = DateTime.Now, TotalLogEntries = logs.Count, Duration = logs.LastOrDefault()?.Timestamp - logs.FirstOrDefault()?.Timestamp, Summary = logs.GroupBy(l => l.Message.Contains("Error") ? "Errors" : "Success") .ToDictionary(g => g.Key, g => g.Count()), DetailedLogs = logs }; var json = JsonConvert.SerializeObject(report, Formatting.Indented); File.WriteAllText($"import_report_{DateTime.Now:yyyyMMdd_HHmmss}.json", json); }
