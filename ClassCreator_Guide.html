Overview

The ClassCreator is a powerful code generation engine that dynamically creates .NET classes, components, and complete application architectures from entity structures. It supports modern development patterns including microservices, cloud functions, frontend components, and database integrations.

??? Code Generation Engine

ClassCreator transforms entity metadata into production-ready code using templates, Roslyn compilation, and modern architectural patterns. It generates everything from simple POCOs to complete enterprise solutions including APIs, validation, documentation, and cloud deployments.

Usage Examples

Complete Application Generation

// Initialize ClassCreator var classCreator = new ClassCreator(dmeEditor); // Define entities var entities = new List<EntityStructure> { customerEntity, productEntity, orderEntity }; // 1. Generate domain models foreach (var entity in entities) { // POCO classes var pocoCode = classCreator.CreatePOCOClass( entity.EntityName, entity, "using System;\nusing System.ComponentModel.DataAnnotations;", null, null, @"C:\Generated\Models\", "MyApp.Models" ); // Validators var validatorCode = classCreator.GenerateFluentValidators( entity, @"C:\Generated\Validators\", "MyApp.Validators" ); // Repositories var repoCode = classCreator.GenerateRepositoryImplementation( entity, @"C:\Generated\Repositories\", "MyApp.Repositories" ); } // 2. Generate database components var dbContextCode = classCreator.GenerateDbContext( entities, "MyApp.Data", @"C:\Generated\Data\" ); // 3. Generate API layer var apiControllers = classCreator.GenerateWebApiControllers( "MainDatabase", entities, @"C:\Generated\Controllers\", "MyApp.Api.Controllers" ); // 4. Generate frontend components foreach (var entity in entities) { var blazorComponent = classCreator.GenerateBlazorComponent( entity, @"C:\Generated\Components\", "MyApp.Web.Components" ); } // 5. Generate cloud functions foreach (var entity in entities) { var azureFunctions = classCreator.GenerateServerlessFunctions( entity, @"C:\Generated\Functions\", CloudProviderType.Azure ); } Console.WriteLine("Complete application generated successfully!");

Advanced Template Usage

// Custom template generation var classCreator = new ClassCreator(dmeEditor); // Define custom template with business logic string customTemplate = @" private :FIELDTYPE _:FIELDNAMEValue; public :FIELDTYPE :FIELDNAME { get => _:FIELDNAMEValue; set { if (_:FIELDNAMEValue != value) { ValidateProperty(value, nameof(:FIELDNAME)); _:FIELDNAMEValue = value; OnPropertyChanged(); } } }"; // Custom using statements string usingStatements = @" using System; using System.ComponentModel; using System.ComponentModel.DataAnnotations; using System.Runtime.CompilerServices;"; // Custom implementations string implementations = "INotifyPropertyChanged, IDataErrorInfo"; // Custom extra code string extraCode = @" public event PropertyChangedEventHandler PropertyChanged; protected virtual void OnPropertyChanged([CallerMemberName] string propertyName = null) { PropertyChanged?.Invoke(this, new PropertyChangedEventArgs(propertyName)); } protected virtual void ValidateProperty(object value, string propertyName) { // Custom validation logic here var context = new ValidationContext(this) { MemberName = propertyName }; Validator.ValidateProperty(value, context); } public string Error => string.Empty; public string this[string columnName] { get { try { var property = GetType().GetProperty(columnName); var value = property?.GetValue(this); var context = new ValidationContext(this) { MemberName = columnName }; Validator.ValidateProperty(value, context); return null; } catch (ValidationException ex) { return ex.Message; } } }"; // Generate with custom template var customClassCode = classCreator.CreateClassFromTemplate( customerEntity.EntityName, customerEntity, customTemplate, usingStatements, implementations, extraCode, @"C:\Generated\Custom\", "MyApp.CustomModels" ); Console.WriteLine("Custom template class generated with advanced validation!");
